# requirements-ui.txt — the Streamlit console (app/), which reaches the API over HTTP.
#
# Installed by Dockerfile.streamlit and, through requirements-dev.txt, by CI and the
# development environment. The API image (Dockerfile.hf) does not install it: the API
# never imports these packages (tests/test_streamlit_image.py checks both).
#
#   pip install -c constraints.txt -r requirements-ui.txt
#
# 1.67 is the first release that allows Pillow 13, which fixes 37 Pillow advisories
# no 12.x release does; 1.62 dropped GitPython (33 advisories) as a dependency
# (Decision 104).
streamlit>=1.67.0,<2.0.0
# Streamlit's tables (st.dataframe, the Tie-out tab) load pyarrow, and
# pyarrow 26 refuses to import beside numpy 1.x ("pyarrow requires NumPy 2.0
# or newer"), which requirements.txt pins. 24.x is the build this was tested with.
pyarrow>=14.0.0,<25.0.0
# The console's HTTP client (app/components/*).
requests>=2.31.0,<3.0.0
